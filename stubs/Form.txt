<?php

namespace RockForms;

use ProcessWire\WireMail;

class {name} extends RockForm
{
  // to disable CSRF for this form
  // to load CSRF on pageload use "pageload" instead of false
  // note that you must only use "pageload" on pages that are not cached!
  // or use 'domready' to load the csrf via ajax on pageload (supports cached pages)
  // const CSRF = false;

  // to disable HTMX for this form
  // const HTMX = false;

  public function buildForm()
  {
    $form = $this;
    $form->setRockFormsRenderer('UIkit');

    // use this to show all hidden fields (like csrf, honeypots, etc)
    // $form->debug();

    $form->addText("forename", "Enter your first name")
      ->setRequired("We need your first name to show it!");
    $f = $form->addText("surname", "Enter your given name")
      ->getLabelPrototype();
    // bd($f);
    $form->addSubmit("submit", "Submit your name");
  }

  /**
   * processInput will be executed on every form submission
   * even if the form has errors or is spam
   */
  public function processInput()
  {
    if ($this->isSpam) {
      // ...
      return;
    }

    if ($this->hasErrors()) {
      // ...
      return;
    }

    // get values without system fields (csrf, honeypot)
    // if you want to get system fields as well use $this->values(true)
    $values = $this->values();

    // you can throw custom errors like this:
    if ($values->surname === 'baz') {
      $this
        ->getComponent('surname')
        ->addError('--- name baz not allowed ---');
    }
  }

  public function processSuccess() {
    $values = $this->values();

    // you can save entries to the database like this
    $fullname = "{$values->forename} {$values->surname}";
    $this->saveEntry("Form submitted by $fullname");

    // add url to rendered table in email
    $values->url = $this->wire->page->url;

    // or you could send an email using wiremail
    $mail = new WireMail();
    $mail->from("robot@yourdomain.com");
    $mail->subject("Form Submission [$this->name]: $fullname");
    $mail->to("you@yourdomain.com");
    $mail->bodyHTML($this->renderTable($values));
    $mail->send();
  }

  public function renderSuccess($entityValues)
  {
    // NOTE: $entityValues are safe to output in your HTML
    return "<div class='uk-alert uk-alert-success'>Thank you for submitting the form, {$entityValues->forename}!</div>";
  }
}
